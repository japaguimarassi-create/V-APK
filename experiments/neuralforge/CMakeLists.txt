cmake_minimum_required(VERSION 3.16)
project(neuralforge_response_stats LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
add_library(response_stats SHARED cpp/response_stats.cpp)
target_include_directories(response_stats PUBLIC cpp)
if(MSVC)
  target_compile_options(response_stats PRIVATE /O2 /W4)
else()
  target_compile_options(response_stats PRIVATE -O3 -Wall -Wextra -Wpedantic)
endif()
set_target_properties(response_stats PROPERTIES OUTPUT_NAME response_stats)
